cmake_minimum_required(VERSION 3.16)
project(rccl_upstream_tests LANGUAGES CXX)
find_package(GTest CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(rccl_bitops ../test/BitOpsTests.cpp)
target_compile_features(rccl_bitops PRIVATE cxx_std_17)
target_include_directories(rccl_bitops PRIVATE ../src/include)
target_link_libraries(rccl_bitops PRIVATE GTest::gtest_main Threads::Threads)
enable_testing()
include(GoogleTest)
gtest_discover_tests(rccl_bitops)
