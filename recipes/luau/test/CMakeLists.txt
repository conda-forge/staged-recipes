cmake_minimum_required(VERSION 3.16)
project(luau_test LANGUAGES CXX)
find_package(Luau CONFIG REQUIRED)
foreach(_c Common Ast Compiler Config Analysis EqSat CodeGen VM Require)
  if(NOT TARGET Luau::${_c})
    message(FATAL_ERROR "Luau::${_c} missing")
  endif()
endforeach()
add_executable(luau_test test.cpp)
target_link_libraries(luau_test PRIVATE Luau::VM Luau::Compiler Luau::CodeGen)
